// WHY THIS EXISTS: `agent-view.spec.ts:261` ("all five E1 mechanics manifests match
// their byte-stable fixture") is ALREADY RED on pristine main — it dies at :266 on the

// would notice a `posting.lossStakes` regression — NEVER EXECUTE on main today.
//
// This probe reproduces :267/:268 EXACTLY, but over the five ids the red assertion

// does retiring `posting.lossStakes` from the derivation break the fixture compare?
import { readFile } from 'node:fs/promises';
import path from 'node:path';
import { expect, test } from '@playwright/test';
import { deriveMechanicsManifest, type MechanicsManifest } from '../src/agent/MechanicsManifest';

const EXPECTED_IDS = ['the-claim', 'e1-dry-gulch', 'e1-night-shift', 'e1-twin-banks', 'e1-baron'];

test('s1354 probe: fixture byte-compare, reached rather than short-circuited', async () => {
  const fixtures = JSON.parse(
    await readFile(path.resolve('e2e/fixtures/e1-mechanics-manifests.json'), 'utf8'),
  ) as MechanicsManifest[];
  const manifests = EXPECTED_IDS.map(deriveMechanicsManifest);

  const derivedKeys = [...new Set(manifests.map((m) => Object.keys(m.posting).join('|')))];
  const fixtureHasLossStakes = JSON.stringify(fixtures).includes('lossStakes');
  console.log(`PROBE derived posting keys: ${JSON.stringify(derivedKeys)}`);
  console.log(`PROBE fixture still carries lossStakes: ${fixtureHasLossStakes}`);
  console.log(`PROBE stringify equal: ${JSON.stringify(manifests) === JSON.stringify(fixtures.slice(0, 5))}`);

  expect(JSON.stringify(manifests)).toBe(JSON.stringify(fixtures.slice(0, 5)));
});
