import { expect, test, type Page } from '@playwright/test';
import { PROFILE_KEY, TOWN_NAME_KEY, profileDataKey, type ProfileState } from '../src/game/ProfileStorage';

// s1332 drain probe — screenshots the e1-drill-yard board card at real card scale.
// Navigation copied verbatim from e2e/board-card-images.spec.ts (the acceptance test).
test('drill yard board card screenshot', async ({ page }, testInfo) => {
  const errors: string[] = [];
  page.on('console', (message) => message.type() === 'error' && errors.push(message.text()));
  page.on('pageerror', (error) => errors.push(error.message));
  await seedProfile(page);
  await page.goto('/');
  await page.evaluate(() => history.replaceState(null, '', '/?debug&epoch=epoch-10-deepsky'));
  await page.getByTestId('start-menu-enter-town').click();
  await page.waitForFunction(() => (window.__GR_TOWN_DIAGNOSTICS__?.frame ?? 0) > 10);
  const target = await page.evaluate((id) => {
    const diagnostics = window.__GR_TOWN_DIAGNOSTICS__!;
    const slot = diagnostics.plaza.slots.find((candidate) => candidate.id === id);
    return slot?.approach ?? slot?.position;
  }, 'tavern');
  if (!target) throw new Error('tavern is absent from town plaza diagnostics');
  for (let step = 0; step < 48; step += 1) {
    if ((await page.evaluate(() => window.__GR_TOWN_DIAGNOSTICS__?.activePrompt)) === 'tavern') break;
    const position = await page.evaluate(() => window.__GR_TOWN_DIAGNOSTICS__!.player);
    const keys: string[] = [];
    if (Math.abs(target.x - position.x) > 0.6) keys.push(target.x > position.x ? 'KeyD' : 'KeyA');
    if (Math.abs(target.z - position.z) > 0.6) keys.push(target.z > position.z ? 'KeyS' : 'KeyW');
    for (const key of keys) await page.keyboard.down(key);
    await page.waitForTimeout(160);
    for (const key of keys.reverse()) await page.keyboard.up(key);
  }
  await expect.poll(() => page.evaluate(() => window.__GR_TOWN_DIAGNOSTICS__?.activePrompt), { timeout: 8_000 }).toBe('tavern');
  await page.getByTestId('town-open-board').click();

  await page.getByTestId('contract-chapter-tab-epoch-1-frontier').click();
  const card = page.getByTestId('contract-card-e1-drill-yard');
  await expect(card).toBeVisible();
  const art = page.getByTestId('contract-art-e1-drill-yard').locator('img');
  await expect(art).toHaveAttribute('src', /e1-drill-yard/);

  // Prove the bytes actually decoded — a broken <img> is still "visible".
  const decoded = await art.evaluate((node) => {
    const image = node as HTMLImageElement;
    return { complete: image.complete, w: image.naturalWidth, h: image.naturalHeight, src: image.src };
  });
  expect(decoded.complete).toBe(true);
  expect(decoded.naturalWidth ?? decoded.w).toBeGreaterThan(0);
  testInfo.annotations.push({ type: 'decoded', description: JSON.stringify(decoded) });
  console.log(`[s1332] ${testInfo.project.name} decoded=${JSON.stringify(decoded)}`);

  const slug = testInfo.project.name;
  await card.screenshot({ path: `reviews/shots-drill-yard-plate/card-${slug}.png` });
  await page.screenshot({ path: `reviews/shots-drill-yard-plate/board-${slug}.png` });
  expect(errors).toEqual([]);
});

async function seedProfile(page: Page): Promise<void> {
  await page.addInitScript(({ profileKey, townKey }) => {
    localStorage.clear();
    sessionStorage.clear();
    const profile: ProfileState = {
      version: 2,
      activeId: 'robin',
      profiles: [{ id: 'robin', name: 'Robin', createdAt: 1, updatedAt: 1, difficultyPreset: 'trail', hintsSeen: [] }],
    };
    localStorage.setItem(profileKey, JSON.stringify(profile));
    localStorage.setItem(townKey, 'Quartz Hill');
  }, { profileKey: PROFILE_KEY, townKey: profileDataKey('robin', TOWN_NAME_KEY) });
}
