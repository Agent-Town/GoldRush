async function seedTown(page: Page, epochId: string): Promise<void> {
  await page.addInitScript(({ activeEpochKey, epochId, firstClaimKey, flatEpochKey, flatFirstClaimKey, flatTownKey, hints, marker, profileKey, townKey }) => {
    if (localStorage.getItem(marker)) return;
    localStorage.clear();
    sessionStorage.clear();
    const profile: ProfileState = {
      version: 2,
      activeId: 'robin',
      profiles: [{ id: 'robin', name: 'Robin', createdAt: 1, updatedAt: 1, difficultyPreset: 'trail', hintsSeen: hints }],
    };
    localStorage.setItem(profileKey, JSON.stringify(profile));
    localStorage.setItem(townKey, 'Quartz Hill');
    localStorage.setItem(flatTownKey, 'Quartz Hill');
    localStorage.setItem(firstClaimKey, '1');
    localStorage.setItem(flatFirstClaimKey, '1');
    localStorage.setItem(activeEpochKey, epochId);
    localStorage.setItem(flatEpochKey, epochId);
    localStorage.setItem(marker, '1');
  }, {
    activeEpochKey: profileDataKey('robin', ACTIVE_EPOCH_KEY),
    epochId,
    firstClaimKey: profileDataKey('robin', FIRST_CLAIM_DONE_KEY),
    flatEpochKey: ACTIVE_EPOCH_KEY,
    flatFirstClaimKey: FIRST_CLAIM_DONE_KEY,
    flatTownKey: TOWN_NAME_KEY,
    hints: OLD_STORY_HINTS,
    marker: `aging2-seeded:${epochId}`,
    profileKey: PROFILE_KEY,
    townKey: profileDataKey('robin', TOWN_NAME_KEY),
  });
}

async function dismiss(page: Page): Promise<void> {
  await page.mouse.click(6, 6);
  await expect(page.getByTestId('story-beat-card')).toHaveCount(0);
}

for (const item of CASES) {
  test(`plain boot: ${item.beat} speaks as ${item.speaker}`, async ({ page }, testInfo) => {
    test.setTimeout(120_000);
    const errors = watchErrors(page);
    await seedTown(page, item.epoch);
    await page.goto('/');
    expect(page.url()).not.toContain('debug');
    await page.waitForFunction(() => Boolean(window.__GR_STORY__));
    const signal: RuntimeStorySignal = { type: 'contract-unlocked', contractId: item.contractId, contractName: 'probe', ledgerBlurb: 'probe' };
    await page.evaluate((value) => window.__GR_STORY__?.emit(value), signal);

    for (let attempt = 0; attempt < 24; attempt += 1) {
      await expect.poll(() => page.evaluate(() => window.__GR_STORY__?.active() ?? null), { timeout: 8_000 }).not.toBeNull();
      if ((await page.evaluate(() => window.__GR_STORY__?.active())) === item.beat) break;
      await dismiss(page);
    }
    const card = page.getByTestId('story-beat-card');
    await expect(card).toHaveAttribute('data-beat-id', item.beat);
    await expect(card).toHaveAttribute('data-speaker', item.speaker);
    await expect(page.locator('.story-beat-card__speaker')).toHaveText(item.name);

    const portrait = page.getByTestId('story-beat-portrait');
    const shape = await portrait.evaluate((node) => {
      const img = node as HTMLImageElement;
      return { src: img.getAttribute('src') ?? '', naturalWidth: img.naturalWidth, objectPosition: getComputedStyle(img).objectPosition };
    });
    expect(shape.src).toContain(item.portrait);
    expect(shape.naturalWidth).toBeGreaterThan(0);
    expect(shape.objectPosition).toBe('50% 42%');

    await mkdir(SHOTS, { recursive: true });
    await page.screenshot({ path: `${SHOTS}/plainboot-${testInfo.project.name}-${item.speaker}.png` });
    expectNoConsoleErrors(errors);
  });
}

test('every registered speaker portrait decodes', async ({ page }) => {
  test.setTimeout(180_000);
  const errors = watchErrors(page);
  await seedTown(page, 'epoch-1-frontier');
  await page.goto('/');
  // The URLs must come from the PAGE's own module graph: `portraitUrl` is built from
  // `import.meta.url`, so the Node-side value is a file:// path the browser cannot fetch. Resolving
  // it here through the dev server's own module is the only honest read (vite DEV, not preview).
  const result = await page.evaluate(async () => {
    const mod = await import('/src/story/speakers.ts');
    const speakers = Object.values(mod.STORY_SPEAKERS) as { id: string; portraitUrl: string }[];
    const bad: string[] = [];
    await Promise.all(speakers.map(({ id, portraitUrl }) => new Promise<void>((resolve) => {
      const img = new Image();
      img.onload = () => { if (!img.naturalWidth) bad.push(id); resolve(); };
      img.onerror = () => { bad.push(id); resolve(); };
      img.src = portraitUrl;
    })));
    return { count: speakers.length, bad, sample: speakers[0]?.portraitUrl ?? '' };
  });
  expect(Object.values(STORY_SPEAKERS).length).toBe(87);
  expect(result.count).toBe(87);
  expect(result.sample.startsWith('http')).toBe(true);
  expect(result.bad).toEqual([]);
  expectNoConsoleErrors(errors);
});
